rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Regras para teste/apresentação do Deas Finance.
    // Permite acesso somente para usuários logados no Firebase Authentication.
    // Isso evita o erro: Missing or insufficient permissions.
    match /{document=**} {
      allow read, write: if request.auth != null;
    }
  }
}
